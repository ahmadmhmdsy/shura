# خطة تنفيذ «شورى» — MVP (المرحلتان 0 + 1)

> **للمنفذين الآليين:** مهارة مطلوبة: superpowers:subagent-driven-development (الموصى بها) أو superpowers:executing-plans لتنفيذ هذه الخطة مهمةً بمهمة. الخطوات بصيغة checkbox (- [ ]).

**الهدف:** بناء تطبيق «شورى» MVP: دردشة محلية بإرسال متوازٍ لأربعة مزودين (OpenRouter، Gemini، MiniMax، OpenCode Go) مع بث مباشر، حفظ مزدوج (Dexie + Markdown)، ومؤشر كونتيكست.

**المعمارية:** واجهة Vite+React+TS داخل غلاف Tauri 2 اختياري؛ طبقة نقل موحدة (fetch المتصفح أو plugin-http داخل Taurي لحل CORS)؛ سجل مزودين واحد فوق AI SDK؛ Dexie مصدر الحقيقة أثناء التشغيل وكاتب Markdown للمرآة الدائمة.

**الحزمة:** React 19، TypeScript strict، Tailwind v4 + shadcn/ui، Zustand، Dexie، Vercel AI SDK v5، Vitest، Tauri 2.

**المرجع:** docs/superpowers/specs/2026-09-05-shura-app-design.md (القرارات D1–D9).

**بيئة التنفيذ:** Windows، Node 24، pnpm 11، جذر المشروع E:\js_projects\شورى (مستودع Git جاهز، أول التزام 0f4d58a). كل الأوامر تعمل من جذر المشروع ما لم يُذكر خلاف ذلك.

export function newId(prefix = ""): string {
  const uuid = typeof crypto !== "undefined" && "randomUUID" in crypto ? crypto.randomUUID() : Math.random().toString(36).slice(2) + Date.now().toString(36);
  return prefix ? prefix + "_" + uuid : uuid;

export function listProviderIds(): ProviderId[] { return Object.keys(PROVIDERS) as ProviderId[]; }

export function getProvider(id: ProviderId): ProviderConfig {
  const p = PROVIDERS[id];
  if (!p) throw new Error(`مزود غير معروف: ${String(id)}`);
  return p;
}

export function computeCost(usage: Pick<Usage, "inputTokens" | "outputTokens" | "cachedInputTokens">, model: Pick<ModelInfo, "pricing">): number | null {
  if (!model.pricing) return null;
  const { inputPerMTok, outputPerMTok } = model.pricing;
  return (usage.inputTokens / 1e6) * inputPerMTok + (usage.outputTokens / 1e6) * outputPerMTok;
}

export type FetchLike = typeof fetch;

export function buildLanguageModel(providerId: ProviderId, apiKey: string, modelId: string, fetchImpl: FetchLike, extraHeaders?: Record<string, string>): LanguageModel {
  if (!apiKey.trim()) throw new Error(`مفتاح API للمزود ${getProvider(providerId).label} غير مضبوط — أضفه من الإعدادات`);
  const p = getProvider(providerId);
  const headers = { ...(p.defaultHeaders ?? {}), ...(extraHeaders ?? {}) };
  if (p.kind === "google") {
    const g = createGoogleGenerativeAI({ apiKey, baseURL: p.baseURL, headers, fetch: fetchImpl });
    return g(modelId);
  }
  const oc = createOpenAICompatible({ name: providerId, baseURL: p.baseURL, apiKey, headers, fetch: fetchImpl });
  return oc(modelId);
}

import type { BuiltPrompt, Turn } from "@/types";

export interface TurnTargetInput { targetId: string; providerId: string; modelId: string; }

const CUSTOM_SEPARATOR = "\n\n[تعليمات خاصة لهذا النموذج]\n";

export function buildTurnTargets(
  turn: Turn,
  targets: readonly TurnTargetInput[],
  syncMode: boolean,
): { prompts: BuiltPrompt[]; dropped: string[] } {
  const shared = turn.sharedPrompt?.trim() || null;

  if (syncMode) {
    if (!shared) throw new Error("اكتب سؤالًا موحدًا أولاً — وضع الربط مُفعّل ويُرسل نفس السؤال للجميع");
    return {
      prompts: targets.map((t) => ({ targetId: t.targetId, providerId: t.providerId as BuiltPrompt["providerId"], modelId: t.modelId, text: shared, hasShared: true, hasCustom: false })),
      dropped: [],
    };
  }

  const prompts: BuiltPrompt[] = [];
  const dropped: string[] = [];
  for (const t of targets) {
    const custom = turn.customs[t.targetId]?.trim() || null;
    if (!shared && !custom) { dropped.push(t.targetId); continue; }
    const text = custom && shared ? shared + CUSTOM_SEPARATOR + custom : (custom ?? shared!);
    prompts.push({ targetId: t.targetId, providerId: t.providerId as BuiltPrompt["providerId"], modelId: t.modelId, text, hasShared: shared !== null, hasCustom: custom !== null });
  }
  if (prompts.length === 0) throw new Error("اكتب سؤالًا — لا يوجد موحد ولا مخصص لأي نموذج");
  return { prompts, dropped };
}

import type { ChatMessage } from "@/types";

export function estimateTokens(text: string): number {
  return text.length === 0 ? 0 : Math.ceil(text.length / 4);
}

export function estimateSessionTokens(messages: readonly Pick<ChatMessage, "text">[]): number {
  return messages.reduce((sum, m) => sum + estimateTokens(m.text), 0);
}

export type ContextLevel = "ok" | "warn" | "danger";

export function contextPercent(used: number, limit: number): number {
  if (limit <= 0) return 0;
  return Math.min(100, Math.round((used / limit) * 100));
}

export function contextLevel(used: number, limit: number): ContextLevel {
  const pct = contextPercent(used, limit);
  if (pct >= 95) return "danger";
  if (pct >= 80) return "warn";
  return "ok";
}

import type { UsageEvent } from "@/types";

export interface UsageSummary {
  inputTokens: number; outputTokens: number; cachedInputTokens: number; totalTokens: number;
  costUSD: number; toolCalls: number; calls: number; errors: number;
}

export function summarize(events: readonly UsageEvent[], filter?: { fromTs?: number }): UsageSummary {
  const s: UsageSummary = { inputTokens: 0, outputTokens: 0, cachedInputTokens: 0, totalTokens: 0, costUSD: 0, toolCalls: 0, calls: 0, errors: 0 };
  for (const e of events) {
    if (filter?.fromTs && e.ts < filter.fromTs) continue;
    s.calls += 1;
    s.inputTokens += e.inputTokens;
    s.outputTokens += e.outputTokens;
    s.cachedInputTokens += e.cachedInputTokens;
    s.costUSD += e.costUSD ?? 0;
    s.toolCalls += e.toolCalls;
    if (e.status === "error") s.errors += 1;
  }
  s.totalTokens = s.inputTokens + s.outputTokens;
  return s;
}

export function groupByProvider(events: readonly UsageEvent[]): Record<string, UsageSummary> {
  const out: Record<string, UsageSummary> = {};
  for (const e of events) {
    (out[e.providerId] ??= { inputTokens: 0, outputTokens: 0, cachedInputTokens: 0, totalTokens: 0, costUSD: 0, toolCalls: 0, calls: 0, errors: 0 });
    const g = out[e.providerId];
    g.calls += 1;
    g.inputTokens += e.inputTokens;
    g.outputTokens += e.outputTokens;
    g.cachedInputTokens += e.cachedInputTokens;
    g.costUSD += e.costUSD ?? 0;
    g.toolCalls += e.toolCalls;
    if (e.status === "error") g.errors += 1;
    g.totalTokens = g.inputTokens + g.outputTokens;
  }
  return out;
}

export interface FileSink {
  ensureDir(path: string): Promise<void>;
  writeFile(path: string, content: string): Promise<void>;
  readFile(path: string): Promise<string>;
  exists(path: string): Promise<boolean>;
  readDir(path: string): Promise<string[]>;
}

export function memoryFileSink(): FileSink & { reset(): void; tree(): string[] } {
  const files = new Map<string, string>();
  return {
    async ensureDir() {},
    async writeFile(p, c) { files.set(p, c); },
    async readFile(p) { const c = files.get(p); if (c === undefined) throw new Error("ENOENT: " + p); return c; },
    async exists(p) { return files.has(p); },
    async readDir() { return []; },
    reset() { files.clear(); },
    tree() { return [...files.keys()]; },
  };
}

  status: Record<string, string>;
  errors: Record<string, string>;

const ILLEGAL = /[<>:"/\\|?*\x00-\x1f]/g;

export function sanitizeTitle(title: string): string {
  return title.replace(ILLEGAL, "-").replace(/\s+/g, "-").replace(/-{2,}/g, "-").replace(/^-+|-+$/g, "").slice(0, 40);
}

export function sessionFolderName(title: string, id: string, createdAt: number): string {
  const d = new Date(createdAt || Date.now());
  const ymd = [d.getFullYear(), String(d.getMonth() + 1).padStart(2, "0"), String(d.getDate()).padStart(2, "0")].join("-");
  const base = sanitizeTitle(title) || "بدون-عنوان";
  return `${ymd}_${base}_${id.slice(0, 8)}`;
}

export function usageRow(u: Usage | null): string {
  if (!u) return "| الاستهلاك | غير متوفر |\n";
  const cost = u.costUSD == null ? "—" : "$" + u.costUSD.toFixed(6);
  return `| المدخلات | ${u.inputTokens} |\n| المخرجات | ${u.outputTokens} |\n| الكاش | ${u.cachedInputTokens} |\n| التكلفة | ${cost} |\n`;
}

function pad2(n: number): string { return String(n).padStart(2, "0"); }

export function turnDirName(turnIndex: number, createdAt: number): string {
  const d = new Date(createdAt || Date.now());
  const t = [d.getHours(), d.getMinutes(), d.getSeconds()].map(pad2).join("");
  return `t${pad2(turnIndex + 1)}_${t}`;
}

export function resultFileName(index: number, providerId: string, modelId: string): string {
  return `${pad2(index + 1)}_${providerId}__${modelId.replace(/[^\w.-]/g, "_")}.md`;
}

export async function writeSessionMeta(sink: FileSink, dir: string, meta: SessionMeta, turns: unknown[] = []): Promise<void> {
  await sink.ensureDir(dir);
  await sink.writeFile(dir + "/meta.json", JSON.stringify({ meta, turns }, null, 2));

export async function appendTurn(sink: FileSink, dir: string, meta: SessionMeta, data: TurnFileData): Promise<void> {
  await sink.ensureDir(dir);
  const existing = (await sink.exists(dir + "/session.md")) ? await sink.readFile(dir + "/session.md") : "";
  const metaJson = (await sink.exists(dir + "/meta.json")) ? JSON.parse(await sink.readFile(dir + "/meta.json")) : { meta, turns: [] };
  const turnIndex = metaJson.turns.length;
  const tdir = `turns/${turnDirName(turnIndex, data.createdAt)}`;

  let section = `\n\n## تمرّة ${turnIndex + 1} — ${new Date(data.createdAt).toLocaleString("ar")}\n`;
  if (data.turn.sharedPrompt) section += `\n> **السؤال الموحد:** ${data.turn.sharedPrompt}\n`;
  for (const p of data.prompts.filter((p) => p.hasCustom)) section += `\n> **خاص بـ ${p.modelId}:** ${p.text}\n`;
  section += "\n";
  for (const r of data.results) {
    const prompt = data.prompts.find((p) => p.targetId === r.targetId)!;
    const st = data.status[r.targetId] ?? "done";
    section += `\n### ${prompt.providerId} — ${prompt.modelId}\n`;
    if (st === "error") section += `\n⛔ ${data.errors[r.targetId] ?? "خطأ"}\n`;
    else section += `\n${r.text}\n`;
    section += "\n" + usageRow(r.usage);
  }

  await sink.writeFile(dir + "/session.md", existing + section);
  await sink.ensureDir(`${dir}/${tdir}`);
  if (data.turn.sharedPrompt) await sink.writeFile(`${dir}/${tdir}/00_shared-prompt.md`, data.turn.sharedPrompt + "\n");
  for (const [i, p] of data.prompts.entries()) {
    if (!p.hasCustom) continue;
    await sink.writeFile(`${dir}/${tdir}/${resultFileName(i, p.providerId, p.modelId)}`, "## سؤال مخصص\n\n" + p.text + "\n");
  }
  for (const [i, r] of data.results.entries()) {
    const prompt = data.prompts.find((p) => p.targetId === r.targetId)!;
    const body = data.status[r.targetId] === "error" ? `⛔ ${data.errors[r.targetId] ?? "خطأ"}` : r.text;
    await sink.writeFile(`${dir}/${tdir}/${resultFileName(i, prompt.providerId, prompt.modelId)}`, body + "\n\n---\n" + usageRow(r.usage));
  }
  metaJson.turns.push({ ...data, dirName: tdir });
  await sink.writeFile(dir + "/meta.json", JSON.stringify(metaJson, null, 2));
}

import { buildLanguageModel, computeCost, getProvider } from "@/lib/providers/registry";
import { getTransportFetch } from "@/lib/transport/transport";
import { mapProviderError } from "@/lib/providers/keyTest";
import type { BuiltPrompt, ProviderId, Usage } from "@/types";

export interface DispatchOutcome { targetId: string; ok: boolean; text: string; error?: string; usage?: Usage; ms: number; }

export interface DispatchArgs {
  sessionId: string; turnId: string;
  prompts: readonly BuiltPrompt[];
  apiKeyFor(providerId: ProviderId): string;
  sessionHeaderFor(providerId: ProviderId): Record<string, string> | undefined;
  timeoutMs?: number;
  fetchImpl?: typeof fetch;
  onUpdate?(targetId: string, chunk: string): void;
  externalSignalFor?(targetId: string): AbortSignal | undefined;
}

export async function dispatchTurn(args: DispatchArgs): Promise<DispatchOutcome[]> {
  const timeout = args.timeoutMs ?? 120_000;
  const results = await Promise.all(args.prompts.map(async (p): Promise<DispatchOutcome> => {
    const t0 = Date.now();
    const ac = new AbortController();
    args.externalSignalFor?.(p.targetId)?.addEventListener("abort", () => ac.abort(new Error("أُلغي بواسطة المستخدم")));
    const timer = setTimeout(() => ac.abort(new Error("انتهت مهلة النداء")), timeout);
    try {
      const model = buildLanguageModel(p.providerId as ProviderId, args.apiKeyFor(p.providerId as ProviderId), p.modelId, args.fetchImpl ?? getTransportFetch(), args.sessionHeaderFor(p.providerId as ProviderId));
      const result = streamText({ model, prompt: p.text, abortSignal: ac.signal });
      let text = "";
      for await (const chunk of result.textStream) { text += chunk; args.onUpdate?.(p.targetId, chunk); }
      const u = await result.usage;
      const info = getProvider(p.providerId as ProviderId).staticModels.find((m) => m.id === p.modelId);
      const usage: Usage = {
        inputTokens: u.inputTokens ?? 0, outputTokens: u.outputTokens ?? 0, cachedInputTokens: u.cachedInputTokens ?? 0,
        costUSD: info ? computeCost({ inputTokens: u.inputTokens ?? 0, outputTokens: u.outputTokens ?? 0, cachedInputTokens: u.cachedInputTokens ?? 0 }, info) : null,
      };
      return { targetId: p.targetId, ok: true, text, usage, ms: Date.now() - t0 };
    } catch (e) {
      return { targetId: p.targetId, ok: false, text: "", error: ac.signal.aborted && !ac.signal.reason ? "أُلغي النداء" : mapProviderError(e), ms: Date.now() - t0 };
    } finally {
      clearTimeout(timer);
    }
  }));
  return results;
}

  const outcomes = await dispatchTurn({
    sessionId: session.id, turnId: turn.id, prompts: built.prompts,
    apiKeyFor: args.apiKeyFor, sessionHeaderFor: args.sessionHeaderFor, timeoutMs: args.timeoutMs,
    externalSignalFor: args.externalSignalFor,
    onUpdate: (targetId, chunk) => cb.onDelta(targetId, chunk),
  });

  for (const o of outcomes) {
    const msg = assistantMsgs.find((m) => m.targetId === o.targetId)!;
    const prompt = built.prompts.find((p) => p.targetId === o.targetId)!;
    const status: ChatMessage["status"] = o.ok ? "done" : "error";
    const final: ChatMessage = { ...msg, text: o.text, status, error: o.error, usage: o.usage };
    await appendMessage(final);
    await recordUsage({ id: newId("u"), sessionId: session.id, turnId: turn.id, ts: Date.now(), providerId: o.ok ? prompt.providerId : prompt.providerId, modelId: prompt.modelId, inputTokens: o.usage?.inputTokens ?? 0, outputTokens: o.usage?.outputTokens ?? 0, cachedInputTokens: o.usage?.cachedInputTokens ?? 0, costUSD: o.usage?.costUSD ?? null, toolCalls: 0, durationMs: o.ms, status: o.ok ? "ok" : "error" });
    if (o.ok && o.usage) {
      const info = getProvider(prompt.providerId).staticModels.find((m) => m.id === prompt.modelId);
      void info ? computeCost(o.usage, info ?? {}) : null;
    }
    cb.onStatus(o.targetId, status, o.error);
    if (dir) {
      await appendTurn(sink, dir, session, {
        turn, prompts: built.prompts.map((p) => ({ targetId: p.targetId, providerId: p.providerId, modelId: p.modelId, text: p.text, hasShared: p.hasShared, hasCustom: p.hasCustom })),
        results: outcomes.map((oo) => ({ targetId: oo.targetId, text: oo.text, usage: oo.usage ?? null })),
        status: Object.fromEntries(outcomes.map((oo) => [oo.targetId, oo.ok ? "done" : "error"])),
        errors: Object.fromEntries(outcomes.map((oo) => [oo.targetId, oo.error ?? ""])),
        createdAt: Date.now(),
      });
    }
  }
  if (dir) await updateSession(session.id, { onDisk: isFilePersistActive() });
  return assistantMsgs;
}

import { useMemo, useRef, useState } from "react";

import { Composer } from "./Composer";
import { TurnView } from "./TurnView";
import { useSessionsStore } from "@/state/sessionsStore";
import { useComposerStore, type ComposerTarget } from "@/state/composerStore";
import { useSettingsStore } from "@/state/settingsStore";
import { estimateSessionTokens } from "@/lib/tokens/estimate";
import { getProvider } from "@/lib/providers/registry";

import { ar } from "@/i18n/ar";
import type { ChatMessage, SessionMeta, Turn } from "@/types";

export function ChatScreen({ session }: { session: SessionMeta }) {
  const { messages, appendMessageLocal, patchMessage } = useSessionsStore();
  const { targets, syncMode, sharedPrompt, customs } = useComposerStore();
  const settings = useSettingsStore();
  const [sending, setSending] = useState(false);
  const aborts = useRef(new Map<string, AbortController>());
  const sessionTokens = useMemo(() => estimateSessionTokens(messages), [messages]);

  const turns = useMemo(() => {
    const map = new Map<string, { prompts: ChatMessage[]; results: ChatMessage[] }>();
    for (const m of messages) {
      const t = map.get(m.turnId) ?? { prompts: [], results: [] };
      (m.role === "user" ? t.prompts : t.results).push(m);
      map.set(m.turnId, t);
    }
    return [...map.entries()];
  }, [messages]);

  const customPrompts = Object.fromEntries(targets.map((t) => [t.targetId, (customs[t.targetId] ?? "").trim() || null]));

  async function doSend(): Promise<void> {
    if (sending) return;
    const turn: Turn = { id: newId("turn"), sharedPrompt: sharedPrompt, customs: { ...customs } };
    await persistTurn(turn);
  }

  async function persistTurn(turn: Turn, retryTargetId?: string, baseResults?: ChatMessage[]): Promise<void> {
    const signalTargets = retryTargetId ? targets.filter((t) => t.targetId === retryTargetId) : targets;
    const turnForTargets: Turn = retryTargetId ? { ...turn, customs: { [retryTargetId]: (customs[retryTargetId] ?? "") } } : turn;
    setSending(true);
    try {
      await sendTurn(
        {
          session, turn: turnForTargets, targets: signalTargets.map((t: ComposerTarget) => ({ targetId: t.targetId, providerId: t.providerId, modelId: t.modelId })),
          syncMode: retryTargetId ? false : syncMode,
          apiKeyFor: (pid) => settings.providerKeys[pid] ?? "",
          sessionHeaderFor: (pid) => (pid === "opencodego" ? { "x-opencode-session": session.id } : undefined),
          externalSignalFor: (tid) => { const ac = new AbortController(); aborts.current.set(tid, ac); return ac.signal; },
          timeoutMs: settings.timeoutMs,
        },
        {
          onDelta: (tid, chunk) => { const m = baseResults?.find((x) => x.targetId === tid); if (m) patchMessage(m.id, { text: m.text + chunk, status: "streaming" }); else { const mm = messages[messages.length - 1]; if (mm?.targetId === tid) patchMessage(mm.id, { text: mm.text + chunk, status: "streaming" }); } },
          onStatus: (tid, st, err) => {
            const list = baseResults ?? useSessionsStore.getState().messages;
            const m = [...list].reverse().find((x) => x.targetId === tid && x.role === "assistant");
            if (m) patchMessage(m.id, { status: st, error: err });
            if (st === "error" && err) toast.error(err);
          },
          onDropped: (ids) => { if (ids.length) toast.warning(ar.droppedTargets + " (" + ids.length + ")"); },
        },
      );
    } finally { setSending(false); aborts.current.clear(); }
  }

  async function onToggleLayout(): Promise<void> {
    const next = session.layoutMode === "vertical" ? "horizontal" : "vertical";
    await updateSession(session.id, { layoutMode: next });
    useSessionsStore.setState((s) => ({ sessions: s.sessions.map((x) => (x.id === session.id ? { ...x, layoutMode: next } : x)) }));
  }

  if (targets.length === 0 && messages.length === 0) {
    return (
      <main className="flex-1 flex flex-col h-full min-w-0">
        <div className="flex-1 grid place-items-center text-sura-muted text-sm">أضف نماذج من (+) ثم اكتب سؤالك.</div>
        <Composer onSend={() => void doSend()} onStop={() => {}} onExtractSummary={() => {}} sending={sending} sessionTokens={sessionTokens} contextLimit={128000} />
      </main>
    );
  }

  return (
    <main className="flex-1 flex flex-col h-full min-w-0">
      <div className="flex-1 overflow-y-auto p-4 space-y-4" dir="rtl">
        {turns.map(([turnId, t]) => (
          <TurnView key={turnId} prompts={t.prompts} results={t.results} layout={session.layoutMode}
            customPrompts={customPrompts} onToggleLayout={() => void onToggleLayout()}
            onStop={(tid) => { aborts.current.get(tid)?.abort(new Error("أُلغي بواسطة المستخدم")); patchMessageByTarget(tid, "stopped"); }}
            onRetry={(tid) => {
              const lastTurn = turns[turns.length - 1]?.[1];
              const m = lastTurn.results.find((r) => r.targetId === tid);
              if (m) void persistTurn({ id: m.turnId, sharedPrompt: null, customs: {} }, tid, [m]);
            }} />
        ))}
        {turns.length === 0 && <p className="text-sura-muted text-sm">لا رسائل بعد.</p>}
      </div>
      <Composer onSend={() => void doSend()} onStop={() => aborts.current.forEach((a) => a.abort(new Error("أُلغي بواسطة المستخدم")))}
        onExtractSummary={() => {}} sending={sending}
        sessionTokens={sessionTokens}
        contextLimit={getProvider(targets[0]?.providerId ?? "openrouter").staticModels.find((m) => m.id === targets[0]?.modelId)?.contextLength ?? 128000} />
    </main>
  );
}

function patchMessageByTarget(targetId: string, status: ChatMessage["status"]): void {
  const s = useSessionsStore.getState();
  const m = [...s.messages].reverse().find((x) => x.targetId === targetId && x.role === "assistant");
  if (m) s.patchMessage(m.id, { status });
}

import { ensureDir, exists, readDir, readFile, writeFile } from "@tauri-apps/plugin-fs";
import type { FileSink } from "./sink";

export function tauriFileSink(): FileSink {
  return {
    async ensureDir(p) { await ensureDir(p, { recursive: true }); },
    async writeFile(p, content) { await writeFile(p, content, { create: true }); },
    async readFile(p) { return new TextDecoder().decode(await readFile(p)); },
    async exists(p) { return exists(p); },
    async readDir(p) { return (await readDir(p)).map((e) => e.name); },
  };
}
